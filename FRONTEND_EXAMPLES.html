api.js - Helper functions

// Save the token after login
function saveToken(token) {
  localStorage.setItem("auth_token", token);
}

function getToken() {
  return localStorage.getItem("auth_token");
}

function clearToken() {
  localStorage.removeItem("auth_token");
}

// Make API calls with auth header
async function apiCall(endpoint, options = {}) {
  const token = getToken();
  const headers = {
    "Content-Type": "application/json",
    ...options.headers
  };
  if (token) {
    headers.Authorization = `Bearer ${token}`;
  }
  
  const response = await fetch(`http://127.0.0.1:5000/api${endpoint}`, {
    ...options,
    headers
  });
  
  if (!response.ok) {
    const error = await response.json();
    throw new Error(error.error || `HTTP ${response.status}`);
  }
  
  return response.json();
}

// Redirect to login if not authenticated
function requireAuth() {
  if (!getToken()) {
    window.location.href = "/login.html";
  }
}

4️⃣ Student Activity (Do Questions)

student/activity.html

<!DOCTYPE html>
<html>
<head>
  <title>Learning Activity</title>
  <style>
    body { font-family: Arial; background: #f5f5f5; padding: 20px; }
    .container { max-width: 700px; margin: 0 auto; }
    .card { background: white; padding: 20px; margin: 15px 0; border-radius: 8px; box-shadow: 0 2px 8px rgba(0,0,0,0.1); }
    .question { border-left: 4px solid #3498db; padding: 15px; margin: 15px 0; }
    input[type="text"], select { width: 100%; padding: 10px; margin: 10px 0; border: 1px solid #ddd; border-radius: 4px; }
    button { background: #27ae60; color: white; padding: 12px 20px; border: none; border-radius: 4px; cursor: pointer; }
    button:hover { background: #229954; }
    .feedback { padding: 15px; margin: 10px 0; border-radius: 4px; }
    .correct { background: #d4edda; color: #155724; }
    .incorrect { background: #f8d7da; color: #721c24; }
    .hint { background: #fff3cd; color: #856404; padding: 10px; margin: 10px 0; border-radius: 4px; }
  </style>
</head>
<body>
  <div class="container">
    <div class="card">
      <h1 id="title"></h1>
      <p id="instructions"></p>
      <div id="explanation"></div>
    </div>

    <div id="questionsContainer"></div>

    <div class="card">
      <button onclick="submitAnswers()">Submit Answers</button>
      <button onclick="goBack()">Back to Dashboard</button>
    </div>

    <div id="feedback"></div>
  </div>

  <script src="../api.js"></script>
  <script>
    requireAuth();
    let activity = null;
    let startTime = {};

    async function loadActivity() {
      try {
        const params = new URLSearchParams(window.location.search);
        const id = params.get("id");
        activity = await apiCall(`/activities/${id}`);
        
        document.getElementById("title").textContent = activity.title;
        document.getElementById("instructions").textContent = activity.content.instructions;
        
        const exp = activity.content.explanation;
        let expHtml = `<h3>${exp.text}</h3><p>${exp.example}</p>`;
        if (exp.steps && exp.steps.length) {
          expHtml += "<ol>" + exp.steps.map(s => `<li>${s}</li>`).join("") + "</ol>";
        }
        document.getElementById("explanation").innerHTML = expHtml;

        let qHtml = "";
        activity.content.questions.forEach((q, idx) => {
          startTime[q.id] = Date.now();
          qHtml += `
            <div class="card question">
              <h3>Question ${idx + 1}</h3>
              <p>${q.prompt}</p>
          `;
          
          if (q.type === "mcq") {
            qHtml += `<select id="q_${q.id}"><option value="">Choose...</option>`;
            q.options.forEach(opt => {
              qHtml += `<option value="${opt}">${opt}</option>`;
            });
            qHtml += "</select>";
          } else {
            qHtml += `<input type="text" id="q_${q.id}" placeholder="Your answer">`;
          }
          
          if (q.hint) {
            qHtml += `<div class="hint"><strong>Hint:</strong> ${q.hint}</div>`;
          }
          
          qHtml += "</div>";
        });
        
        document.getElementById("questionsContainer").innerHTML = qHtml;
      } catch (err) {
        alert("Error loading activity: " + err.message);
      }
    }

    async function submitAnswers() {
      try {
        const answers = activity.content.questions.map(q => {
          const elem = document.getElementById(`q_${q.id}`);
          return {
            question_id: q.id,
            response: elem.value,
            response_time: (Date.now() - startTime[q.id]) / 1000,
            attempts: 1,
            hint_used: false
          };
        });

        const result = await apiCall(`/activities/${activity.id}/submit`, {
          method: "POST",
          body: JSON.stringify({ answers })
        });

        let fbHtml = `
          <div class="card">
            <h2>Your Results</h2>
            <p><strong>Score:</strong> ${result.summary.score}</p>
            <p><strong>Accuracy:</strong> ${result.summary.accuracy}%</p>
            <p><strong>Mastery:</strong> ${result.summary.current_mastery}%</p>
            <p><strong>Trend:</strong> ${result.summary.trend}</p>
            <p><em>${result.summary.message}</em></p>
            <button onclick="window.location.href='dashboard.html'">Back to Dashboard</button>
          </div>
        `;
        document.getElementById("feedback").innerHTML = fbHtml;
        document.getElementById("questionsContainer").innerHTML = "";
      } catch (err) {
        alert("Error submitting: " + err.message);
      }
    }

    function goBack() {
      window.location.href = "dashboard.html";
    }

    loadActivity();
  </script>
</body>
</html>

6️⃣ Teacher Reviews (Approve/Modify/Reject)

teacher/reviews.html

<!DOCTYPE html>
<html>
<head>
  <title>Adaptation Reviews</title>
  <style>
    body { font-family: Arial; background: #f5f5f5; padding: 20px; }
    .container { max-width: 900px; margin: 0 auto; }
    .card { background: white; padding: 20px; margin: 15px 0; border-radius: 8px; box-shadow: 0 2px 8px rgba(0,0,0,0.1); border-left: 4px solid #3498db; }
    .reason { background: #ecf0f1; padding: 15px; margin: 15px 0; border-radius: 6px; }
    .recommendation { background: #d4edda; padding: 15px; margin: 15px 0; border-radius: 6px; }
    button { padding: 8px 12px; margin: 5px; border: none; border-radius: 4px; cursor: pointer; }
    .approve { background: #27ae60; color: white; }
    .modify { background: #f39c12; color: white; }
    .reject { background: #c0392b; color: white; }
    .question { padding: 10px; background: #f9f9f9; margin: 10px 0; border-radius: 4px; border-left: 3px solid #3498db; }
  </style>
</head>
<body>
  <div class="container">
    <h1>Pending Adaptation Reviews</h1>
    <p>These are automatically generated adaptations based on student performance. Review and approve before they reach students.</p>
    <div id="reviews"></div>
  </div>

  <script src="../api.js"></script>
  <script>
    requireAuth();

    async function loadReviews() {
      try {
        const reviews = await apiCall("/reviews?status=pending");
        
        let html = "";
        for (const card of reviews) {
          const student = card.student;
          const rec = card.recommended_adaptation;
          const reason = card.adaptation_reason;
          
          html += `
            <div class="card">
              <h2>${student.name} - ${card.topic}</h2>
              <p><strong>Cycle:</strong> ${card.cycle}</p>
              
              <div class="reason">
                <h3>Why this adaptation?</h3>
                <p>${reason}</p>
              </div>
              
              <div class="recommendation">
                <h3>Recommended Changes</h3>
                <ul>
                  <li><strong>Difficulty:</strong> ${rec.difficulty}</li>
                  <li><strong>Scaffolding:</strong> ${rec.scaffolding}</li>
                  <li><strong>Questions:</strong> ${rec.question_count}</li>
                  <li><strong>Visual Support:</strong> ${rec.visual_support ? "Yes" : "No"}</li>
                  <li><strong>Format:</strong> ${rec.content_format}</li>
                  <li><strong>Practice:</strong> ${rec.practice_type}</li>
                </ul>
              </div>
              
              <h3>Questions Preview</h3>
          `;
          
          card.content.questions.slice(0, 3).forEach((q, idx) => {
            html += `
              <div class="question">
                <strong>Q${idx+1}:</strong> ${q.prompt}
                <br><small>Answer: ${q.answer} | Hint: ${q.hint || "None"}</small>
              </div>
            `;
          });
          
          if (card.content.questions.length > 3) {
            html += `<p><em>...and ${card.content.questions.length - 3} more questions</em></p>`;
          }
          
          html += `
            <div style="margin-top: 15px;">
              <button class="approve" onclick="approve(${card.activity_id})">✓ Approve</button>
              <button class="modify" onclick="modify(${card.activity_id})">✎ Modify</button>
              <button class="reject" onclick="reject(${card.activity_id})">✕ Reject</button>
            </div>
            <hr>
          </div>
          `;
        }
        
        document.getElementById("reviews").innerHTML = html || "<p>No pending reviews!</p>";
      } catch (err) {
        alert("Error loading reviews: " + err.message);
      }
    }

    async function approve(id) {
      if (confirm("Approve this activity?")) {
        try {
          await apiCall(`/reviews/${id}/approve`, { method: "POST", body: JSON.stringify({}) });
          alert("Approved! Student will see this now.");
          loadReviews();
        } catch (err) {
          alert("Error: " + err.message);
        }
      }
    }

    async function modify(id) {
      const changes = prompt("Enter JSON changes (e.g., {\"difficulty\": \"beginner\", \"question_count\": 5}):");
      if (changes) {
        try {
          await apiCall(`/reviews/${id}/modify`, {
            method: "POST",
            body: JSON.stringify({
              changes: JSON.parse(changes),
              approve: confirm("Approve after modifying?")
            })
          });
          alert("Modified and regenerated!");
          loadReviews();
        } catch (err) {
          alert("Error: " + err.message);
        }
      }
    }

    async function reject(id) {
      const reason = prompt("Why are you rejecting this?");
      if (reason) {
        try {
          await apiCall(`/reviews/${id}/reject`, {
            method: "POST",
            body: JSON.stringify({ reason })
          });
          alert("Rejected.");
          loadReviews();
        } catch (err) {
          alert("Error: " + err.message);
        }
      }
    }

    loadReviews();
  </script>
</body>
</html>
